  1. 5/10第一次寫 micro:bit 程式：自己會跑 A～Z 跑馬燈、會發出聲音。
  2. 5/13循線小車：一直調直線和轉彎的速度，找出最快又不會跑出線的數字。
  3. 5/14～5/17實驗一、三：數字變成亂碼，加了 basic.pause(200) 讓資料排隊；實驗三改了傳出去的名字。

實驗一：夜盲症測試

0921 最新版・MakeCode Python　看實驗 →

量 5 次、算平均，用無線電傳到電腦。

  • 雷達只讀 8×8 裡的 (3,3) 一格，除以 10 換成公分
  • 雷達和超音波合成一行「R1:xx U:yy」，每次等 0.5 秒
  • 號碼牌 is_running：遙控器只舉牌，主迴圈才做實驗
打開完整程式（75 行）
# 1. 定義全域變數
total_radar = 0
total_ultra = 0
is_running = False

# 2. 將「實驗執行邏輯」獨立出來
def run_experiment():
    global total_radar, total_ultra
    total_radar = 0
    total_ultra = 0
    basic.show_icon(IconNames.TARGET)

    # 宣告實驗開始
    radio.send_string("--- EXP 1 ---")
    basic.pause(500)

    for index in range(5):
        # 讀取【光學雷達】數據
        dist_radar = matrixLidarDistance.matrix_point_output(matrixLidarDistance.Addr.ADDR4, 3, 3) / 10

        # 讀取【超音波】數據 (提醒：若數值卡死不動，請將 P13 與 P14 的線對調)
        dist_ultra = maqueenPlusV2.read_ultrasonic(DigitalPin.P13, DigitalPin.P14)

        # 【神級優化】將雷達與超音波合併成一行發送，絕對不會超過 19 字元極限！
        msg = "R" + str(index + 1) + ":" + str(dist_radar) + " U:" + str(dist_ultra)
        radio.send_string(msg)
        basic.pause(500)

        total_radar += dist_radar
        total_ultra += dist_ultra

    # 傳送平均值 (合併發送)
    avg_msg = "RA:" + str(total_radar / 5) + " UA:" + str(total_ultra / 5)
    radio.send_string(avg_msg)
    basic.pause(500)

    basic.show_icon(IconNames.YES)


# 3. 無線電遙控接收區：只發號碼牌，不直接執行！
def on_received_string(receivedString):
    global is_running
    if receivedString == "GO":
        is_running = True
radio.on_received_string(on_received_string)


# 4. 手動按鍵觸發 (一樣發號碼牌，讓流程統一更穩定)
def on_button_pressed_a():
    global is_running
    is_running = True
input.on_button_pressed(Button.A, on_button_pressed_a)


# ==========================================
# 主迴圈：小車在這邊不斷檢查號碼牌並執行動作
# ==========================================
def on_forever():
    global is_running
    # 如果看到號碼牌舉起來了，才開始做實驗
    if is_running == True:
        run_experiment()
        is_running = False    # 執行完畢，放下號碼牌
basic.forever(on_forever)

實驗四：生死一瞬間

0921 最新版・MakeCode Python　看實驗 →

用三種速度往前衝，15 公分內看到障礙物就煞車，量滑了多遠。

  • current_speed 改成 150、205、255，就是 20、30、40 cm/s
  • 煞車那一刻記一次距離，等 1.5 秒再記一次，相減
  • 傳回 Speed、Trig、Slide 三個數字
打開完整程式（95 行）
# --- 實驗四：生死一瞬間 (優化統一格式版) ---

# 1. 在這裡設定測試速度的 PWM 值！(150, 205, 或 255)
current_speed = 150

# 2. 預定緊急煞車的基準線：設定在 15 公分
brake_distance = 15

# 【新增】號碼牌機制
is_running = False

def run_braking_test():
    global current_speed, brake_distance

    basic.show_icon(IconNames.TARGET)

    # 宣告實驗四開始，格式與前三實驗統一
    radio.send_string("--- EXP 4 ---")
    basic.pause(500)

    # 廣播目前設定的速度 (精簡為 Speed:150)
    radio.send_string("Speed:" + str(current_speed))
    basic.pause(1000) # 給測量者 1 秒鐘的準備時間

    # 【起步】：雙輪同時啟動，以設定速度前進 (ALL_MOTOR 可以簡化程式)
    maqueenPlusV2.control_motor(maqueenPlusV2.MyEnumMotor.ALL_MOTOR, maqueenPlusV2.MyEnumDir.FORWARD, current_speed)

    # 【自動駕駛監控迴圈】
    while True:
        # 讀取光學雷達前方距離
        dist = matrixLidarDistance.matrix_point_output(matrixLidarDistance.Addr.ADDR4, 3, 3) / 10

        # 煞車判斷：若距離小於等於 15 公分，且大於 0
        if dist > 0 and dist <= brake_distance:
            # 1. 瞬間切斷動力
            maqueenPlusV2.control_motor_stop(maqueenPlusV2.MyEnumMotor.ALL_MOTOR)

            # 2. 記錄煞車瞬間的精準距離
            trigger_dist = dist

            # 3. 讓車輛因物理慣性滑行 1.5 秒，確保完全靜止
            basic.pause(1500)

            # 4. 重新讀取完全靜止後的「最終距離」
            final_dist = matrixLidarDistance.matrix_point_output(matrixLidarDistance.Addr.ADDR4, 3, 3) / 10

            # 5. 計算實際滑行距離
            slide_dist = trigger_dist - final_dist

            # 6. 【關鍵修改】：改用 send_string，並加入等待時間防塞車
            radio.send_string("Trig:" + str(trigger_dist))
            basic.pause(500)
            radio.send_string("Slide:" + str(slide_dist))
            basic.pause(500)

            basic.show_icon(IconNames.NO)
            break # 跳出監控，等待下一次測試

        basic.pause(10) # 每 10 毫秒極速監測一次

    basic.pause(2000)
    basic.show_icon(IconNames.YES)

    global is_running

        is_running = True

實驗五-A：窄山路

0921 最新版（V16 三段變速）・MakeCode Python　看實驗 →

看到路障就繞過去；同一支程式可以切換雷達或超音波。

  • get_safe_dist()：預警距離用實驗四的數字
  • 三種速度各有一組後退、轉出、通過、轉回的時間
  • 遙控器 A 用雷達、B 用超音波
打開完整程式（135 行）
# --- 實驗五-A：動態轉向閃避版 (V16 三段變速獨立校正版) ---

# 🌟 設定區
# ==========================================
# 【請在這裡手動修改 current_speed 進行三種速度測試】
# 速度 150：約 20cm/s
# 速度 205：約 30cm/s
# 速度 255：約 40cm/s
# ==========================================
current_speed = 255

sensor_mode = 1      # 1: 光學雷達, 2: 超音波
is_running = False   # 號碼牌機制

# 🌟 根據實驗數據設定的防護罩
def get_safe_dist(speed):
    if speed == 150: return 13.3
    elif speed == 205: return 14.6
    else: return 17.0

# 🌟 自動轉向閃避動作 (三段變速獨立校正版)
def auto_dodge():
    global is_running
    radio.send_string("Action: DODGE!")

    # === 【關鍵新增】：根據目前速度，自動切換對應的時間參數 ===
    # 先設定預設變數
    back_t = 200      # 1. 稍微後退的時間
    turn_out_t = 200  # 2. 向右轉向閃避的時間
    pass_t = 900      # 3. 直行越過路障的時間
    turn_in_t = 100   # 4. 轉回正前方的時間
    fwd_t = 1000      # 5. 轉正後，往前開的安全時間

    if current_speed == 150:
        # 🐢 低速參數 (速度慢，需要的時間比較長，請依實際狀況微調)
        back_t = 200
        turn_out_t = 360
        pass_t = 500
        turn_in_t = 400
        fwd_t = 1000
    elif current_speed == 205:
        # 🚗 中速參數 (你原本設定的預設值)
        back_t = 200
        turn_out_t = 280
        pass_t = 400
        turn_in_t = 380
        fwd_t = 800
    elif current_speed == 255:
        # 🚀 高速參數 (速度極快，只要轉/走一下下就到了，請依實際狀況微調)
        back_t = 200
        turn_out_t = 200
        pass_t = 300
        turn_in_t = 280
        fwd_t = 600
    # ========================================================

    # 1. 稍微後退 (預防太靠近撞到)
    maqueenPlusV2.control_motor(maqueenPlusV2.MyEnumMotor.ALL_MOTOR, maqueenPlusV2.MyEnumDir.BACKWARD, current_speed)
    basic.pause(back_t)

    # 2. 向右轉向閃避 (右馬達減速或反轉)
    maqueenPlusV2.control_motor(maqueenPlusV2.MyEnumMotor.LEFT_MOTOR, maqueenPlusV2.MyEnumDir.FORWARD, current_speed)
    maqueenPlusV2.control_motor(maqueenPlusV2.MyEnumMotor.RIGHT_MOTOR, maqueenPlusV2.MyEnumDir.BACKWARD, current_speed)
    basic.pause(turn_out_t)

    # 3. 直行越過路障
    maqueenPlusV2.control_motor(maqueenPlusV2.MyEnumMotor.ALL_MOTOR, maqueenPlusV2.MyEnumDir.FORWARD, current_speed)
    basic.pause(pass_t)

    # 4. 轉回正前方 (向左轉回)
    maqueenPlusV2.control_motor(maqueenPlusV2.MyEnumMotor.LEFT_MOTOR, maqueenPlusV2.MyEnumDir.BACKWARD, current_speed)
    maqueenPlusV2.control_motor(maqueenPlusV2.MyEnumMotor.RIGHT_MOTOR, maqueenPlusV2.MyEnumDir.FORWARD, current_speed)
    basic.pause(turn_in_t)

    # 5. 轉正後，往前開一段「你指定的安全時間」
    maqueenPlusV2.control_motor(maqueenPlusV2.MyEnumMotor.ALL_MOTOR, maqueenPlusV2.MyEnumDir.FORWARD, current_speed)
    basic.pause(fwd_t)

    # 6. 安全煞停：時間到了立刻停止馬達，避免掉下桌子！
    maqueenPlusV2.control_motor_stop(maqueenPlusV2.MyEnumMotor.ALL_MOTOR)

    # 閃避完成後停止實驗，放下號碼牌
    is_running = False
    radio.send_string("Dodge Done!")


# 🌟 無線電接收設定：只負責修改變數與發號碼牌
def on_received_string(receivedString):
    global sensor_mode, current_speed, is_running

    if receivedString == "A": # 雷達模式
        sensor_mode = 1
        is_running = True
        radio.send_string("--- 5A Lidar ---")
    elif receivedString == "B": # 超音波模式
        sensor_mode = 2
        is_running = True
        radio.send_string("--- 5A Sonar ---")
    # 改變速度
    elif receivedString == "SPD150": current_speed = 150
    elif receivedString == "SPD205": current_speed = 205
    elif receivedString == "SPD255": current_speed = 255
radio.on_received_string(on_received_string)

# 🌟 主迴圈：小車在這邊不斷檢查號碼牌並執行動作
def on_forever():
    global is_running

    # 如果實驗啟動中
    if is_running == True:
        safe_dist = get_safe_dist(current_speed)
        dist = 0

        # 讀取感測距離
        if sensor_mode == 1:
            dist = matrixLidarDistance.matrix_point_output(matrixLidarDistance.Addr.ADDR4, 3, 3) / 10
        else:
            dist = maqueenPlusV2.read_ultrasonic(DigitalPin.P13, DigitalPin.P14)

def get_true_distance(sx, sy, tx, ty):
    q_x = [sx]
    q_y = [sy]
    q_d = [0]
    # 解決行 38 報錯：不使用乘法，直接寫滿 24 個 False
    v_map = [False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False,False]
    v_map[sx + sy * 4] = True
    head = 0

    while head < len(q_x):
        cx = q_x[head]
        cy = q_y[head]
        cd = q_d[head]
        head += 1

        # 找到終點，回傳真實步數
        if cx == tx and cy == ty:
            return cd

        # 檢查四個方向 (北東南西) 尋找暢通活路
        if cy < 5 and not v_map[cx + (cy+1)*4] and is_safe_to_go(cx, cy+1):
            v_map[cx + (cy+1)*4] = True; q_x.append(cx); q_y.append(cy+1); q_d.append(cd+1)
        if cx < 3 and not v_map[(cx+1) + cy*4] and is_safe_to_go(cx+1, cy):
            v_map[(cx+1) + cy*4] = True; q_x.append(cx+1); q_y.append(cy); q_d.append(cd+1)
        if cy > 0 and not v_map[cx + (cy-1)*4] and is_safe_to_go(cx, cy-1):
            v_map[cx + (cy-1)*4] = True; q_x.append(cx); q_y.append(cy-1); q_d.append(cd+1)
        if cx > 0 and not v_map[(cx-1) + cy*4] and is_safe_to_go(cx-1, cy):
            v_map[(cx-1) + cy*4] = True; q_x.append(cx-1); q_y.append(cy); q_d.append(cd+1)

    return 999 # 如果發現四面都是死路，回傳極大值

def do_action(move_type, target_time):

    while time_run < target_time:
